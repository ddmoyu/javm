cmake_minimum_required(VERSION 3.25)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/package.json" JAVM_PACKAGE_JSON)
string(JSON JAVM_APP_VERSION GET "${JAVM_PACKAGE_JSON}" version)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" JAVM_CMAKE_VERSION "${JAVM_APP_VERSION}")

project(JAVM VERSION "${JAVM_CMAKE_VERSION}" LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JAVM_USE_PREBUILT_DEPENDENCIES "Use verified component Releases" OFF)
option(JAVM_CORE_ONLY "Build isolated database/HTTP extension without media or deployment" OFF)
if(WIN32)
    set(JAVM_DEFAULT_TRANSPORT winhttp)
else()
    set(JAVM_DEFAULT_TRANSPORT curl)
endif()
set(JAVM_EXTENSION_TRANSPORT "${JAVM_DEFAULT_TRANSPORT}" CACHE STRING "Extension WebSocket transport")
set_property(CACHE JAVM_EXTENSION_TRANSPORT PROPERTY STRINGS winhttp curl)
if(NOT JAVM_EXTENSION_TRANSPORT MATCHES "^(winhttp|curl)$" OR
   (JAVM_EXTENSION_TRANSPORT STREQUAL "winhttp" AND NOT WIN32))
    message(FATAL_ERROR "Select curl transport on macOS/Linux, or winhttp/curl on Windows.")
endif()
if(WIN32 AND JAVM_EXTENSION_TRANSPORT STREQUAL "curl" AND JAVM_USE_PREBUILT_DEPENDENCIES)
    message(FATAL_ERROR "Published Windows curl components are HTTP-only. Build the isolated core from source.")
endif()
if(JAVM_USE_PREBUILT_DEPENDENCIES)
    if(WIN32)
        set(JAVM_DEPENDENCY_PATHS "${PROJECT_SOURCE_DIR}/build/dependencies/paths.cmake")
    else()
        if(APPLE)
            set(JAVM_COMPONENT_OS macos)
        else()
            set(JAVM_COMPONENT_OS linux)
        endif()
        if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
            set(JAVM_COMPONENT_ARCH arm64)
        else()
            set(JAVM_COMPONENT_ARCH x64)
        endif()
        set(JAVM_DEPENDENCY_PATHS "${PROJECT_SOURCE_DIR}/build/dependencies/${JAVM_COMPONENT_OS}-${JAVM_COMPONENT_ARCH}/paths.cmake")
    endif()
    if(NOT EXISTS "${JAVM_DEPENDENCY_PATHS}")
        message(FATAL_ERROR "Run pnpm offline:setup to download the verified component Releases.")
    endif()
    include("${JAVM_DEPENDENCY_PATHS}")
endif()

# 中文 MSVC 的 UTF-8 输出可能被 CMake 按 ANSI 误读，导致 Ninja 漏记头文件。
# 只修正本机实测的乱码前缀，其他语言继续使用 CMake 的检测结果。
if(MSVC AND CMAKE_GENERATOR MATCHES "Ninja" AND
   CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "^娉ㄦ剰: 鍖呭惈鏂囦欢:")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件: ")
endif()

add_subdirectory(native/database)
if(JAVM_CORE_ONLY)
    enable_testing()
endif()
add_subdirectory(native/probe)
add_subdirectory(native/media)
add_subdirectory(native/http)
add_subdirectory(native/process)
add_subdirectory(native/platform)
add_subdirectory(native/launcher)
add_subdirectory(native/update)
