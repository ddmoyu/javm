if(JAVM_USE_PREBUILT_DEPENDENCIES)
    if(NOT WIN32)
        set(OPENSSL_ROOT_DIR "${JAVM_PREBUILT_CURL}/sdk/openssl")
        set(OPENSSL_USE_STATIC_LIBS TRUE)
    endif()
    # A new component fingerprint must replace a cached SDK from an older release.
    set(CURL_DIR "${JAVM_PREBUILT_CURL}/sdk/lib/cmake/CURL" CACHE PATH "Verified CURL component SDK" FORCE)
    # The database's official WebSocket transport lives in a sibling directory.
    find_package(CURL CONFIG REQUIRED GLOBAL PATHS "${JAVM_PREBUILT_CURL}/sdk/lib/cmake/CURL" NO_DEFAULT_PATH)
    set(JAVM_CURL_LICENSE "${JAVM_PREBUILT_CURL}/licenses/curl.txt")
else()
include(FetchContent)
file(READ "${PROJECT_SOURCE_DIR}/scripts/offline-dependencies.json" JAVM_DEPENDENCIES)
string(JSON CURL_SOURCE_URL GET "${JAVM_DEPENDENCIES}" curl url)
string(JSON CURL_SOURCE_HASH GET "${JAVM_DEPENDENCIES}" curl sha256)
string(JSON CURL_SOURCE_FILE GET "${JAVM_DEPENDENCIES}" curl file)
if(EXISTS "${PROJECT_SOURCE_DIR}/.tmp/offline-dependencies/${CURL_SOURCE_FILE}")
    set(CURL_SOURCE_URL "${PROJECT_SOURCE_DIR}/.tmp/offline-dependencies/${CURL_SOURCE_FILE}")
endif()
set(BUILD_CURL_EXE OFF)
set(BUILD_SHARED_LIBS OFF)
set(BUILD_STATIC_LIBS ON)
set(BUILD_TESTING OFF)
set(BUILD_LIBCURL_DOCS OFF)
set(BUILD_MISC_DOCS OFF)
set(ENABLE_CURL_MANUAL OFF)
if(WIN32)
    set(CURL_USE_SCHANNEL ON)
    set(CURL_USE_OPENSSL OFF)
else()
    set(CURL_USE_SCHANNEL OFF)
    set(CURL_USE_OPENSSL ON)
    set(CURL_CA_FALLBACK ON)
    if(APPLE)
        # OpenSSL 负责 TLS，系统 SecTrust 负责 macOS 信任链校验。
        set(USE_APPLE_SECTRUST ON)
    endif()
endif()
set(CURL_USE_LIBPSL OFF)
set(CURL_USE_LIBSSH2 OFF)
set(CURL_ZLIB OFF CACHE STRING "No optional compression library" FORCE)
set(CURL_BROTLI OFF CACHE STRING "No optional compression library" FORCE)
set(CURL_ZSTD OFF CACHE STRING "No optional compression library" FORCE)
set(USE_NGHTTP2 OFF)
set(CURL_USE_PKGCONFIG OFF)
if(JAVM_EXTENSION_TRANSPORT STREQUAL "curl")
    # HTTP_ONLY 同时关闭 WebSocket；保留 HTTP(S)/WS(S)，显式关闭其余协议。
    set(HTTP_ONLY OFF)
    set(CURL_DISABLE_WEBSOCKETS OFF)
    foreach(protocol DICT FILE FTP GOPHER IMAP IPFS LDAP LDAPS MQTT POP3 RTSP SMB SMTP TELNET TFTP)
        set(CURL_DISABLE_${protocol} ON)
    endforeach()
else()
    set(HTTP_ONLY ON)
endif()
if(CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreaded")
    set(CURL_STATIC_CRT ON)
endif()
FetchContent_Declare(javm_curl URL "${CURL_SOURCE_URL}" URL_HASH "SHA256=${CURL_SOURCE_HASH}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(javm_curl)
set(JAVM_CURL_LICENSE "${javm_curl_SOURCE_DIR}/COPYING")
endif()
add_library(javm_http STATIC src/http_service.cpp)
target_include_directories(javm_http PUBLIC src)
target_link_libraries(javm_http PUBLIC nlohmann_json::nlohmann_json PRIVATE CURL::libcurl)
if(WIN32)
    target_compile_definitions(javm_http PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    target_compile_options(javm_http PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(javm_http PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_link_libraries(javm_database PRIVATE javm_http)
if(JAVM_CORE_ONLY)
    set(JAVM_HTTP_LICENSE_DIR "${CMAKE_BINARY_DIR}/licenses")
else()
    set(JAVM_HTTP_LICENSE_DIR "${PROJECT_SOURCE_DIR}/bin/licenses")
endif()
add_custom_command(TARGET javm_http POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${JAVM_HTTP_LICENSE_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${JAVM_CURL_LICENSE}" "${JAVM_HTTP_LICENSE_DIR}/curl.txt")
