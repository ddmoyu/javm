add_library(javm_platform STATIC src/service.cpp src/credentials.cpp src/loopback.cpp)
target_include_directories(javm_platform PUBLIC src)
target_link_libraries(javm_platform PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
    target_sources(javm_platform PRIVATE src/protocol_windows.cpp)
    target_compile_definitions(javm_platform PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(javm_platform PRIVATE ws2_32)
elseif(APPLE)
    target_sources(javm_platform PRIVATE src/protocol_macos.cpp)
    target_link_libraries(javm_platform PRIVATE "-framework Security" "-framework CoreFoundation" "-framework CoreServices")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
    pkg_check_modules(GIO_UNIX REQUIRED IMPORTED_TARGET gio-unix-2.0)
    target_sources(javm_platform PRIVATE src/protocol_linux.cpp)
    target_link_libraries(javm_platform PRIVATE PkgConfig::SECRET PkgConfig::GIO_UNIX)
endif()
if(MSVC)
    target_compile_options(javm_platform PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(javm_platform PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_link_libraries(javm_database PRIVATE javm_platform)
if(TARGET javm_library_fixture)
    target_link_libraries(javm_library_fixture PRIVATE javm_platform)
endif()
if(TARGET javm_windows_dispatch_check)
    target_link_libraries(javm_windows_dispatch_check PRIVATE javm_platform)
endif()
