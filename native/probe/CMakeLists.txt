include(FetchContent)
if(JAVM_USE_PREBUILT_DEPENDENCIES)
    set(probe_api_include "${JAVM_PREBUILT_MEDIAINFO}/sdk/include")
else()
    # 与现有组件复用相同源码和哈希，仅提取官方 C API 声明；不改变组件构建配方。
    file(READ "${PROJECT_SOURCE_DIR}/scripts/offline-dependencies.json" dependencies)
    string(JSON source_url GET "${dependencies}" mediainfo url)
    string(JSON source_hash GET "${dependencies}" mediainfo sha256)
    string(JSON source_file GET "${dependencies}" mediainfo file)
    if(EXISTS "${PROJECT_SOURCE_DIR}/.tmp/offline-dependencies/${source_file}")
        set(source_url "${PROJECT_SOURCE_DIR}/.tmp/offline-dependencies/${source_file}")
    endif()
    FetchContent_Declare(mediainfo_source URL "${source_url}" URL_HASH "SHA256=${source_hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(mediainfo_source)
    set(probe_api_include "${mediainfo_source_SOURCE_DIR}/MediaInfoLib/Source")
endif()
add_library(javm_probe STATIC src/paths.cpp src/media_info.cpp src/inspect.cpp)
target_link_libraries(javm_probe PUBLIC nlohmann_json::nlohmann_json PRIVATE ${CMAKE_DL_LIBS})
target_include_directories(javm_probe PUBLIC src)
target_include_directories(javm_probe SYSTEM PRIVATE "${probe_api_include}")
target_compile_definitions(javm_probe PRIVATE UNICODE _UNICODE)
if(WIN32)
    target_sources(javm_probe PRIVATE src/capture_windows.cpp)
    target_compile_definitions(javm_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    target_sources(javm_probe PRIVATE src/capture_posix.cpp)
    if(APPLE)
        target_compile_definitions(javm_probe PRIVATE _DARWIN_C_SOURCE)
    endif()
    find_package(Threads REQUIRED)
    target_link_libraries(javm_probe PRIVATE Threads::Threads)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        include(CheckSymbolExists)
        set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
        check_symbol_exists(posix_spawn_file_actions_addclosefrom_np "spawn.h" JAVM_HAVE_SPAWN_CLOSEFROM)
        unset(CMAKE_REQUIRED_DEFINITIONS)
        if(NOT JAVM_HAVE_SPAWN_CLOSEFROM)
            message(FATAL_ERROR "The isolated POSIX probe requires posix_spawn_file_actions_addclosefrom_np (glibc 2.34+).")
        endif()
        target_compile_definitions(javm_probe PRIVATE _GNU_SOURCE)
    endif()
endif()
if(MSVC)
    target_compile_options(javm_probe PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(javm_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

    include(tests.cmake)
endif()
